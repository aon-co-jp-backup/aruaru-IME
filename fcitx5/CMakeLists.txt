cmake_minimum_required(VERSION 3.12)
project(fcitx5-aruaru-ime CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(GNUInstallDirs)
find_package(Fcitx5Core REQUIRED)
add_library(aruaru-ime MODULE aruaru-ime.cpp)
target_link_libraries(aruaru-ime Fcitx5::Core)
set_target_properties(aruaru-ime PROPERTIES PREFIX "")
install(TARGETS aruaru-ime DESTINATION "${CMAKE_INSTALL_LIBDIR}/fcitx5")
install(FILES aruaru-ime-addon.conf RENAME aruaru-ime.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon")
install(FILES aruaru-ime-im.conf RENAME aruaru-ime.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod")
